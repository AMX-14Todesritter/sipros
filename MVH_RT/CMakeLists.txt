cmake_minimum_required(VERSION 3.18)
project(MvhRayTracingStudy LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
option(MVH_RT_BUILD_DATA_EXPORTER "Build the exporter against unchanged CPU MVH code" ON)
option(MVH_RT_BUILD_OPTIX "Build the independent OptiX tutorial (requires SDK headers)" OFF)
if(MVH_RT_BUILD_DATA_EXPORTER)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../mvh" "${CMAKE_BINARY_DIR}/mvh_dependency" EXCLUDE_FROM_ALL)
  add_executable(export_mvh_sample data_export/export_sample.cpp)
  target_link_libraries(export_mvh_sample PRIVATE mvh_original)
  target_compile_options(export_mvh_sample PRIVATE -ffast-math -march=native)
  set_target_properties(export_mvh_sample PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(MVH_RT_BUILD_OPTIX)
  add_subdirectory(optix_example)
endif()
if(MVH_RT_BUILD_DATA_EXPORTER)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME mvh_rt_snapshot_replay
    COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_data.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/test_data")
endif()
