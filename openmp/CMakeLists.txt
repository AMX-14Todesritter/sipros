# Optional validation executable: original main + original startProcessingMvh path,
# stopped immediately after searchDatabaseMvh, before WDP/Xcorr changes the results.
# Only a build-directory copy is instrumented; production sources stay unchanged.
option(SIPROS_MVH_REFERENCE_TEST "Build original-path MVH snapshot executable for validation" OFF)
if(SIPROS_MVH_REFERENCE_TEST)
  file(READ "${PROJECT_SOURCE_DIR}/src/ms2scanvector.cpp" MVH_REFERENCE_SOURCE)
  set(MVH_REFERENCE_ANCHOR "\tsearchDatabaseMvh();\n\n\t// Postprocessing all MS2 scans' results by mult-threading")
  string(FIND "${MVH_REFERENCE_SOURCE}" "${MVH_REFERENCE_ANCHOR}" MVH_REFERENCE_POSITION)
  if(MVH_REFERENCE_POSITION EQUAL -1)
    message(FATAL_ERROR "Original MVH snapshot anchor not found")
  endif()
  string(REPLACE "${MVH_REFERENCE_ANCHOR}"
    "\tsearchDatabaseMvh();\n\twriteOutputMvh((std::filesystem::path(sOutputFile).parent_path() / \"mvh_psms.tsv\").string());\n\treturn;\n\n\t// Postprocessing all MS2 scans' results by mult-threading"
    MVH_REFERENCE_SOURCE "${MVH_REFERENCE_SOURCE}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ms2scanvector_reference.cpp" "${MVH_REFERENCE_SOURCE}")
  set(MVH_REFERENCE_SOURCES ${SRC_LIST})
  list(REMOVE_ITEM MVH_REFERENCE_SOURCES "${PROJECT_SOURCE_DIR}/src/ms2scanvector.cpp")
  add_executable(sipros_mvh_reference ${MVH_REFERENCE_SOURCES}
    "${CMAKE_CURRENT_BINARY_DIR}/ms2scanvector_reference.cpp" "${PROJECT_SOURCE_DIR}/analysis/mvh_search/reference_main.cpp")
  set_target_properties(sipros_mvh_reference PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/ms2scanvector.cpp")
  foreach(MVH_PROPERTY COMPILE_OPTIONS COMPILE_DEFINITIONS LINK_LIBRARIES)
    get_target_property(MVH_PROPERTY_VALUE sipros_mvh_search ${MVH_PROPERTY})
    if(MVH_PROPERTY_VALUE)
      set_property(TARGET sipros_mvh_reference PROPERTY ${MVH_PROPERTY} "${MVH_PROPERTY_VALUE}")
    endif()
  endforeach()
endif()
